    @classmethod
    def setUpClass(cls):
        cls.model = joblib.load(ROOT / "tourism_project/deployment/model.joblib")
        cls.schema = json.loads((ROOT / "tourism_project/deployment/schema.json").read_text())
        cls.X = pd.read_csv(ROOT / "Xtest.csv")

    def test_schema_and_excluded_interactions(self):
        self.assertEqual(self.X.columns.tolist(), self.schema["features"])
        for col in ["CustomerID", "ProdTaken", "DurationOfPitch", "PitchSatisfactionScore",
                    "NumberOfFollowups", "TypeofContact", "ProductPitched"]:
            self.assertNotIn(col, self.schema["features"])

    def test_probabilities(self):
        probs = self.model.predict_proba(self.X.head(10))
        self.assertEqual(probs.shape, (10, 2))
        self.assertTrue(np.isfinite(probs).all())
        np.testing.assert_allclose(probs.sum(axis=1), 1.0, atol=1e-6)
